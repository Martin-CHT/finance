Jeden společný Google Sheet pro všechny moduly. Volitelná synchronizace (záloha transakcí, cílů, investic, majetku) i možnost zálohovat sem AI klíče, aby šly nahrát na jiném zařízení.

Jak nastavit Apps Script (návod + ukázkový kód)
  1. Otevřete prázdný Google Sheet, který bude sloužit jako datové úložiště.
  2. V menu zvolte Rozšíření → Apps Script.
  3. Vložte ukázkový kód níže, uložte projekt.
  4. Klikněte Deploy → New deployment → Web app, nastavte „Execute as: Me", „Who has access: Anyone".
  5. Zkopírujte URL končící /exec a vložte do pole výše.
/** Minimální Apps Script pro Finance Terminal — sjednocený endpoint. */
const SHEET_CONFIG = 'Config';
const TOKEN = 'vlozte_zde_stejny_token_jako_v_nastaveni'; // volitelné

function doGet(e) {
  if (!checkToken(e)) return _json({ok:false, error:'bad token'});
  const action = (e.parameter.action || 'getConfig');
  if (action === 'getConfig') return _json({ok:true, config: readConfig()});
  return _json({ok:false, error:'unknown action'});
}

function doPost(e) {
  if (!checkToken(e)) return _json({ok:false, error:'bad token'});
  const body = JSON.parse(e.postData.contents || '{}');
  if (body.action === 'saveConfig') {
    writeConfig(body.data || {});
    return _json({ok:true});
  }
  if (body.action === 'saveData') {
    // body.module = 'prijmy' | 'sporeni' | 'majetek' | ...
    // body.data   = libovolný JSON
    writeModule(body.module, body.data);
    return _json({ok:true});
  }
  return _json({ok:false, error:'unknown action'});
}

function checkToken(e) {
  if (!TOKEN) return true;
  return (e.parameter.token === TOKEN);
}

function readConfig() {
  const sh = _getSheet(SHEET_CONFIG);
  const rows = sh.getDataRange().getValues();
  const cfg = {};
  rows.forEach(r => { if (r[0]) cfg[r[0]] = r[1]; });
  return cfg;
}

function writeConfig(cfg) {
  const sh = _getSheet(SHEET_CONFIG);
  sh.clear();
  const rows = Object.entries(cfg);
  if (rows.length) sh.getRange(1,1,rows.length,2).setValues(rows);
}

function writeModule(name, data) {
  const sh = _getSheet('mod_' + name);
  sh.clear();
  sh.getRange(1,1).setValue(JSON.stringify(data));
}

function _getSheet(name) {
  const ss = SpreadsheetApp.getActive();
  return ss.getSheetByName(name) || ss.insertSheet(name);
}

function _json(o) {
  return ContentService.createTextOutput(JSON.stringify(o))
    .setMimeType(ContentService.MimeType.JSON);
}
